    foreach(FFMPEG_COMPONENT IN ITEMS avformat.61 avcodec.61 avutil.59)
      file(GLOB FFMPEG_CANDIDATES LIST_DIRECTORIES FALSE
        "${FFMPEG_LIBRARY_DIR}/lib${FFMPEG_COMPONENT}.dylib"
        "${FFMPEG_LIBRARY_DIR}/lib${FFMPEG_COMPONENT}.*.dylib")
      if(NOT FFMPEG_CANDIDATES)
        message(FATAL_ERROR "Qt's bundled lib${FFMPEG_COMPONENT} dylib was not found in ${FFMPEG_LIBRARY_DIR}")
      endif()
      list(SORT FFMPEG_CANDIDATES COMPARE NATURAL ORDER DESCENDING)
      list(GET FFMPEG_CANDIDATES 0 FFMPEG_DYLIB)
      list(APPEND FFMPEG_LIBRARIES "${FFMPEG_DYLIB}")
    endforeach()
